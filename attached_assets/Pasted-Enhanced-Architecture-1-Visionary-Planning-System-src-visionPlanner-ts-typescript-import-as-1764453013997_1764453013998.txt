    issues: string[];
    recommendations: string[];
    score: number;
  }> {

${workspaceStructure}

Look for:
- Missing configuration files
- Security vulnerabilities
- Performance bottlenecks
- Architecture problems
- Best practice violations
- Scalability concerns

Provide specific, actionable feedback.`;

    const response = await this.aiService.chat(
      [{ role: 'user', content: analysisPrompt }],
      'gpt-4o'
    );

    // Parse the response to extract issues and recommendations
    return {
      issues: this.extractItems(response.content, 'ISSUES:'),
      recommendations: this.extractItems(response.content, 'RECOMMENDATIONS:'),
      score: this.calculateHealthScore(response.content)
    };
  }

  private extractItems(content: string, section: string): string[] {
    const sectionIndex = content.indexOf(section);
    if (sectionIndex === -1) return [];
    
    const nextSection = content.indexOf('\n\n', sectionIndex);
    const sectionContent = content.substring(
      sectionIndex + section.length,
      nextSection !== -1 ? nextSection : content.length
    );
    
    return sectionContent
      .split('\n')
      .filter(line => line.trim().startsWith('-'))
      .map(line => line.trim().substring(1).trim());
  }

  private calculateHealthScore(content: string): number {
    // Simple heuristic based on issue density
    const issueCount = (content.match(/issue|problem|error|warning|vulnerability/gi) || []).length;
    const positiveCount = (content.match(/good|excellent|proper|correct|secure/gi) || []).length;
    
    const baseScore = 80;
    const adjustment = Math.min(20, positiveCount * 2) - Math.min(20, issueCount * 5);
    return Math.max(0, Math.min(100, baseScore + adjustment));
  }
}

const ENHANCED_SYSTEM_PROMPT = `You are AI-DAN 2.0, an advanced autonomous AI development partner with visionary planning capabilities.

export class EnhancedAIService extends AIService {
  private visionPlanner: VisionPlanner;

  constructor() {
    super();
    this.visionPlanner = new VisionPlanner(this);
  }

  async enhancedChat(
    messages: Message[], 
    model: string, 
    workspaceContext?: string,
    options: {
      enableVisionPlanning?: boolean;
      analyzeProjectHealth?: boolean;
      generateTemplates?: boolean;
    } = {}
  ): Promise<{
    content: string;
    actions: ParsedAction[];
    vision?: ProjectVision;
    healthAnalysis?: any;
  }> {
    // Enhanced system prompt
    const systemMessage: Message = {
      role: 'system',
      content: ENHANCED_SYSTEM_PROMPT + (workspaceContext ? `\n\nCURRENT WORKSPACE:\n${workspaceContext}` : '')
    };

    const allMessages = [systemMessage, ...messages];

    try {
      // Get AI response
      const response = await this.chat(allMessages, model);
      
      let vision: ProjectVision | undefined;
      let healthAnalysis: any;

      // Generate vision plan for complex projects
      if (options.enableVisionPlanning && this.isComplexProject(messages)) {
        const lastUserMessage = messages.filter(m => m.role === 'user').pop();
        if (lastUserMessage) {
          vision = await this.visionPlanner.createProjectVision(lastUserMessage.content, workspaceContext);
        }
      }

      // Analyze project health if requested
      if (options.analyzeProjectHealth && workspaceContext) {
        healthAnalysis = await this.visionPlanner.analyzeProjectHealth(workspaceContext);
      }

      return {
        content: response.content,
        actions: response.actions,
        vision,
        healthAnalysis
      };
    } catch (error: any) {
      throw new Error(`Enhanced AI request failed: ${error.message}`);
    }
  }

  private isComplexProject(messages: Message[]): boolean {
    const lastMessage = messages.filter(m => m.role === 'user').pop();
    if (!lastMessage) return false;

    const complexKeywords = [
      'game', '3d', 'vr', 'ar', 'multiplayer', 'database', 'authentication',
      'api', 'mobile app', 'desktop app', 'enterprise', 'saas', 'platform'
    ];

    const content = lastMessage.content.toLowerCase();
    return complexKeywords.some(keyword => content.includes(keyword));
  }

  async generateCodeReview(code: string, context: string): Promise<{
    score: number;
    issues: string[];
    suggestions: string[];
    optimizations: string[];
  }> {
    const reviewPrompt = `Perform comprehensive code review:

CODE TO REVIEW:
\`\`\`
${code}
\`\`\`

CONTEXT: ${context}

Analyze for:
1. Code quality and readability
2. Performance implications
3. Security vulnerabilities
4. Architecture and design patterns
5. Best practices adherence
6. Potential bugs and edge cases
7. Optimization opportunities

Provide specific, actionable feedback.`;

    const response = await this.chat(
      [{ role: 'user', content: reviewPrompt }],
      'gpt-4o'
    );

    return this.parseCodeReview(response.content);
  }

  private parseCodeReview(content: string): {
    score: number;
    issues: string[];
    suggestions: string[];
    optimizations: string[];
  } {
    // Parse the AI response to extract structured feedback
    // This is a simplified implementation
    const lines = content.split('\n');
    const issues: string[] = [];
    const suggestions: string[] = [];
    const optimizations: string[] = [];

    let currentSection = '';
    
    lines.forEach(line => {
      if (line.includes('ISSUES:') || line.includes('PROBLEMS:')) {
        currentSection = 'issues';
      } else if (line.includes('SUGGESTIONS:') || line.includes('RECOMMENDATIONS:')) {
        currentSection = 'suggestions';
      } else if (line.includes('OPTIMIZATIONS:') || line.includes('IMPROVEMENTS:')) {
        currentSection = 'optimizations';
      } else if (line.trim().startsWith('-') || line.trim().startsWith('•')) {
        const item = line.trim().substring(1).trim();
        if (currentSection === 'issues') {
          issues.push(item);
        } else if (currentSection === 'suggestions') {
          suggestions.push(item);
        } else if (currentSection === 'optimizations') {
          optimizations.push(item);
        }
      }
    });

    // Calculate score based on issue density
    const score = Math.max(0, 100 - (issues.length * 5));

    return {
      score,
      issues,
      suggestions,
      optimizations
    };
  }
}

        });

        if (selected) {

            value: selected.template.id
          });

          if (projectName) {
            await templateManager.createFromTemplate(selected.template.id, projectName);
          }
        }
      }
    })
  );

  // Enhanced existing commands
  context.subscriptions.push(
    vscode.commands.registerCommand('aidan.enhancedExecuteAction', async () => {
      const action = await vscode.window.showInputBox({

typescript
import * as vscode from 'vscode';
import { EnhancedAIService } from './enhancedAIService';
import { AgentExecutor } from './agentExecutor';
import { FileOperations } from './fileOperations';
import { ProjectTemplateManager } from './projectTemplates';
import { VisionPlanner, ProjectVision } from './visionPlanner';

export class EnhancedChatViewProvider implements vscode.WebviewViewProvider {
  private _view?: vscode.WebviewView;
  private messages: any[] = [];
  private workspaceContext: string = '';
  private currentModel: string = 'gpt-4o';
  private currentVision?: ProjectVision;

  constructor(
    private readonly extensionUri: vscode.Uri,
    private readonly aiService: EnhancedAIService,
    private readonly agentExecutor: AgentExecutor,
    private readonly fileOps: FileOperations,
    private readonly templateManager: ProjectTemplateManager,
    private readonly visionPlanner: VisionPlanner
  ) {}

  resolveWebviewView(webviewView: vscode.WebviewView) {
    this._view = webviewView;

    webviewView.webview.options = {
      enableScripts: true,
      localResourceRoots: [this.extensionUri]
    };

    webviewView.webview.html = this.getEnhancedHtmlContent();

    webviewView.webview.onDidReceiveMessage(async (data) => {
      switch (data.type) {
        case 'sendEnhancedMessage':
          await this.handleEnhancedMessage(data.message, data.options);
          break;
        case 'createFromTemplate':
          await this.templateManager.createFromTemplate(data.templateId, data.projectName);
          break;
        case 'generateVision':
          await this.generateProjectVision(data.prompt);
          break;
        case 'analyzeHealth':
          await this.analyzeProjectHealth();
          break;
        // ... other cases
      }
    });

    this.initializeEnhancedWorkspace();
  }

  private async handleEnhancedMessage(content: string, options: any = {}) {
    this.messages.push({ role: 'user', content });
    this.postMessage({ type: 'userMessage', content });
    this.postMessage({ type: 'thinking', isThinking: true });

    try {
      const response = await this.aiService.enhancedChat(
        this.messages,
        this.currentModel,
        this.workspaceContext,
        {
          enableVisionPlanning: true,
          analyzeProjectHealth: options.analyzeHealth
        }
      );

      this.messages.push({ role: 'assistant', content: response.content });
      
      if (response.vision) {
        this.currentVision = response.vision;
        this.postMessage({
          type: 'projectVision',
          vision: response.vision
        });
      }

      if (response.healthAnalysis) {
        this.postMessage({
          type: 'healthAnalysis',
          analysis: response.healthAnalysis
        });
      }

      this.postMessage({
        type: 'assistantMessage',
        content: response.content,
        actions: response.actions,
        model: this.currentModel,
        hasVision: !!response.vision
      });

      // Auto-execute actions if enabled and this is the first message in a complex project
      if (response.actions && response.actions.length > 0) {
        const config = vscode.workspace.getConfiguration('aidan');
        const autoExecute = config.get<boolean>('autoExecute') || false;
        
        if (autoExecute || this.isInitialProjectSetup()) {
          await this.executeActions(response.actions);
        }
      }
    } catch (error: any) {
      this.postMessage({
        type: 'error',
        message: error.message || 'Failed to get enhanced AI response'
      });
    } finally {
      this.postMessage({ type: 'thinking', isThinking: false });
    }
  }

  private isInitialProjectSetup(): boolean {
    return this.messages.filter(m => m.role === 'user').length <= 2;
  }

  private async generateProjectVision(prompt: string) {
    const vision = await this.visionPlanner.createProjectVision(prompt, this.workspaceContext);
    this.currentVision = vision;
    this.postMessage({
      type: 'projectVision',
      vision
    });
  }

  private async analyzeProjectHealth() {
    const analysis = await this.visionPlanner.analyzeProjectHealth(this.workspaceContext);
    this.postMessage({
      type: 'healthAnalysis',
      analysis
    });
  }

  async executeEnhancedAction(prompt: string) {
    await this.handleEnhancedMessage(prompt, { enableVisionPlanning: true });
  }

  private async initializeEnhancedWorkspace() {
    try {
      this.workspaceContext = await this.fileOps.getWorkspaceStructure();
      
      // Auto-analyze project health on startup
      const config = vscode.workspace.getConfiguration('aidan');
      if (config.get<boolean>('autoAnalyzeOnStartup')) {
        const analysis = await this.visionPlanner.analyzeProjectHealth(this.workspaceContext);
        if (analysis.score < 70) {
          this.postMessage({
            type: 'healthWarning',
            analysis
          });
        }
      }
    } catch {
      // Workspace might not be open
    }
  }

  private getEnhancedHtmlContent(): string {
    // Returns enhanced HTML with vision planning UI, health indicators, template browser
    // Implementation would be similar to original but with additional UI components
    return `<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>AI-DAN 2.0 Enhanced</title>
    <style>
        /* Enhanced styles with vision panels, health indicators, etc. */
        .vision-panel {
            background: var(--vscode-editor-background);
            border: 1px solid var(--vscode-panel-border);
            border-radius: 8px;
            padding: 16px;
            margin: 12px 0;
        }
        .health-indicator {
            display: inline-block;
            padding: 4px 8px;
            border-radius: 12px;
            font-size: 11px;

        }
        .health-good { background: var(--vscode-testing-iconPassed); color: white; }

    </style>

<body>

  private postMessage(message: any) {
    this._view?.webview.postMessage(message);
  }
